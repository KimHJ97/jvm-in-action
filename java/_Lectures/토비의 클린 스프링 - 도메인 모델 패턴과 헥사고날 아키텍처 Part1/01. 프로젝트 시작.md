# 프로젝트 시작


 - `JDK 설치`
    - IDE(IntelliJ)의 JDK 다운로드
    - JDK 배포판 웹사이트에서 다운로드 받아서 설치
    - JDK 버전을 관리하는 도구를 이용해서 설치
        - SDKMAN, jabba, Asdf, jvms
```bash
# SDKMAN 설치
curl -s "https://get.sdkman.io" | bash

# 터미널 사용 초기화
source "$HOME/.sdkman/bin/sdkman-init.sh"

# 설치 확인
sdk version

# 설치 가능한 JDK 목록 조회
sdk list java

# JDK 설치
sdk install java 21.0.6-librca

# 디폴트 JDK: 터미닐 접속시 기본으로 적용되는 JDK 버전 설정
sdk default java 21.0.5-librca

# 현재 터미널 JDK 버전 변경
sdk use java 21.0.5-librca

# 현재 경로에 JDK 기본 버전 설정(.sdkmanrc)
sdk env init

# 환경 정보를 보고 JDK 설치
sdk env install

# 폴더 접속시 env 파일을 보고 자동 JDK 버전 설정
vi ~/.sdkman/etc/config
sdkman_auto_init=true # 해당 값을 true로 변경
```
